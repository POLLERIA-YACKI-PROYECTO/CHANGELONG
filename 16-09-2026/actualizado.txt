-- ============================================================
-- SISTEMA DE GESTIÓN - POLLERÍA DOÑA YACKI
-- Base de datos: polleria_yacki
-- Motor: InnoDB | Charset: utf8mb4 | Collation: utf8mb4_unicode_ci
-- Compatibilidad: MySQL 8.0+
-- ============================================================
-- Descripción: Script de creación completa de la base de datos
--              del sistema de gestión de la pollería.
--              Incluye estructura, índices, llaves foráneas,
--              datos iniciales y verificación final.
-- ============================================================

DROP DATABASE IF EXISTS polleria_yacki;
CREATE DATABASE polleria_yacki
  CHARACTER SET utf8mb4
  COLLATE utf8mb4_unicode_ci;

USE polleria_yacki;

SET FOREIGN_KEY_CHECKS = 0;


-- ============================================================
-- TABLA: categorias
-- ============================================================
-- Propósito: Almacena las categorías de productos de la carta.
-- Uso: Admin (CRUD), cliente (filtro), mesero (navegación).
-- Origen de datos: Carga inicial + panel administrativo.
-- ============================================================
CREATE TABLE categorias (
    id INT PRIMARY KEY AUTO_INCREMENT,
    nombre VARCHAR(50) NOT NULL,
    icono VARCHAR(50) DEFAULT NULL,
    descripcion VARCHAR(255) DEFAULT NULL,
    orden INT DEFAULT 0,
    activo BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    deleted_at TIMESTAMP NULL DEFAULT NULL,

    INDEX idx_categorias_nombre (nombre),
    INDEX idx_categorias_activo (activo),
    INDEX idx_categorias_orden (orden)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;


-- ============================================================
-- TABLA: productos
-- ============================================================
-- Propósito: Catálogo de productos con precio, stock e imagen.
-- Uso: Admin (CRUD), cliente (lectura), mesero (lectura).
-- Origen de datos: Carga inicial + panel administrativo.
-- Relación: categoria_id -> categorias.id
-- ============================================================
CREATE TABLE productos (
    id INT PRIMARY KEY AUTO_INCREMENT,
    categoria_id INT NOT NULL,
    nombre VARCHAR(100) NOT NULL,
    precio DECIMAL(10,2) NOT NULL,
    precio_compra DECIMAL(10,2) DEFAULT 0.00,
    descripcion VARCHAR(255) DEFAULT NULL,
    imagen VARCHAR(255) DEFAULT NULL,
    stock INT DEFAULT 0,
    stock_minimo INT DEFAULT 5,
    unidad_medida VARCHAR(20) DEFAULT 'unidad',
    disponible BOOLEAN DEFAULT TRUE,
    agotado BOOLEAN DEFAULT FALSE,
    destacado BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    deleted_at TIMESTAMP NULL DEFAULT NULL,

    FOREIGN KEY (categoria_id) REFERENCES categorias(id) ON DELETE RESTRICT,

    INDEX idx_productos_categoria (categoria_id),
    INDEX idx_productos_nombre (nombre),
    INDEX idx_productos_disponible (disponible),
    INDEX idx_productos_agotado (agotado),
    INDEX idx_productos_destacado (destacado),
    INDEX idx_productos_precio (precio),
    INDEX idx_productos_stock (stock)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;


-- ============================================================
-- TABLA: usuarios
-- ============================================================
-- Propósito: Personal del sistema (administrativo y operativo).
-- Uso: Login admin, login mesero, gestión de personal.
-- Origen de datos: Carga inicial + panel de personal.
-- Diferencia con clientes: son trabajadores, no consumidores.
-- ============================================================
CREATE TABLE usuarios (
    id INT PRIMARY KEY AUTO_INCREMENT,
    nombre VARCHAR(100) NOT NULL,
    apellido VARCHAR(100) DEFAULT NULL,
    dni VARCHAR(8) UNIQUE NOT NULL,
    email VARCHAR(100) UNIQUE DEFAULT NULL,
    telefono VARCHAR(15) DEFAULT NULL,
    direccion VARCHAR(255) DEFAULT NULL,
    rol ENUM('admin', 'cajero', 'mesero', 'cocinero', 'delivery') DEFAULT 'mesero',
    password VARCHAR(255) NOT NULL,
    foto VARCHAR(255) DEFAULT NULL,
    fecha_contratacion DATE DEFAULT NULL,
    salario DECIMAL(10,2) DEFAULT NULL,
    activo BOOLEAN DEFAULT TRUE,
    ultimo_acceso TIMESTAMP NULL DEFAULT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    deleted_at TIMESTAMP NULL DEFAULT NULL,

    INDEX idx_usuarios_dni (dni),
    INDEX idx_usuarios_email (email),
    INDEX idx_usuarios_rol (rol),
    INDEX idx_usuarios_activo (activo),
    INDEX idx_usuarios_nombre (nombre)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;


-- ============================================================
-- TABLA: clientes
-- ============================================================
-- Propósito: Clientes registrados que piden desde la web.
-- Uso: Login cliente, registro cliente, historial de compras.
-- Origen de datos: Auto-registro desde la carta web.
-- Diferencia con usuarios: son consumidores finales.
-- ============================================================
CREATE TABLE clientes (
    id INT PRIMARY KEY AUTO_INCREMENT,
    nombre VARCHAR(100) NOT NULL,
    apellido VARCHAR(100) DEFAULT NULL,
    dni VARCHAR(8) UNIQUE DEFAULT NULL,
    email VARCHAR(100) UNIQUE DEFAULT NULL,
    password VARCHAR(255) NOT NULL,
    telefono VARCHAR(15) DEFAULT NULL,
    direccion VARCHAR(255) DEFAULT NULL,
    tipo_cliente ENUM('regular', 'frecuente', 'vip') DEFAULT 'regular',
    puntos INT DEFAULT 0,
    activo BOOLEAN DEFAULT TRUE,
    ultimo_acceso TIMESTAMP NULL DEFAULT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    deleted_at TIMESTAMP NULL DEFAULT NULL,

    INDEX idx_clientes_dni (dni),
    INDEX idx_clientes_email (email),
    INDEX idx_clientes_nombre (nombre),
    INDEX idx_clientes_telefono (telefono)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;


-- ============================================================
-- TABLA: mesas
-- ============================================================
-- Propósito: Estado y ubicación de las mesas del local.
-- Uso: Mesero (asignar cliente), admin (supervisión).
-- Origen de datos: Carga inicial + operación del mesero.
-- ============================================================
CREATE TABLE mesas (
    id INT PRIMARY KEY AUTO_INCREMENT,
    numero INT UNIQUE NOT NULL,
    capacidad INT DEFAULT 4,
    ubicacion VARCHAR(50) DEFAULT 'Sala Principal',
    ocupada BOOLEAN DEFAULT FALSE,
    cliente VARCHAR(100) DEFAULT NULL,
    cantidad_personas INT DEFAULT 0,
    hora_ocupacion TIMESTAMP NULL DEFAULT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    deleted_at TIMESTAMP NULL DEFAULT NULL,

    INDEX idx_mesas_numero (numero),
    INDEX idx_mesas_ocupada (ocupada),
    INDEX idx_mesas_ubicacion (ubicacion)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;


-- ============================================================
-- TABLA: pedidos
-- ============================================================
-- Propósito: Pedidos tomados por mesero o cajero en el local.
-- Uso: Mesero (crear), cajero (cobrar), admin (reportes).
-- Origen de datos: Operación del mesero/cajero.
-- Relación: usuario_id -> usuarios, mesa_id -> mesas.
-- Nota: items almacenado como JSON.
-- ============================================================
CREATE TABLE pedidos (
    id INT PRIMARY KEY AUTO_INCREMENT,
    mesa_id INT DEFAULT NULL,
    usuario_id INT NOT NULL,
    cliente_id INT DEFAULT NULL,
    cliente_nombre VARCHAR(100) DEFAULT NULL,
    items JSON NOT NULL,
    subtotal DECIMAL(10,2) DEFAULT 0.00,
    igv DECIMAL(10,2) DEFAULT 0.00,
    total DECIMAL(10,2) NOT NULL,
    tipo ENUM('local', 'delivery', 'paraLlevar') DEFAULT 'local',
    tipo_entrega ENUM('local', 'delivery', 'paraLlevar', 'motorizada') DEFAULT 'local',
    estado ENUM('pendiente', 'preparando', 'listo', 'entregado', 'cancelado') DEFAULT 'pendiente',
    metodo_pago ENUM('efectivo', 'tarjeta', 'yape', 'plin', 'transferencia') DEFAULT NULL,
    pagado BOOLEAN DEFAULT FALSE,
    fecha_pago TIMESTAMP NULL DEFAULT NULL,
    observaciones TEXT DEFAULT NULL,
    hora_inicio TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    hora_preparacion TIMESTAMP NULL DEFAULT NULL,
    hora_entrega TIMESTAMP NULL DEFAULT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    deleted_at TIMESTAMP NULL DEFAULT NULL,

    FOREIGN KEY (mesa_id) REFERENCES mesas(id) ON DELETE SET NULL,
    FOREIGN KEY (usuario_id) REFERENCES usuarios(id) ON DELETE RESTRICT,
    FOREIGN KEY (cliente_id) REFERENCES clientes(id) ON DELETE SET NULL,
SELECT clave, valor FROM configuracion WHERE clave LIKE '%_QR';
    INDEX idx_pedidos_mesa (mesa_id),
    INDEX idx_pedidos_usuario (usuario_id),
    INDEX idx_pedidos_cliente (cliente_id),
    INDEX idx_pedidos_estado (estado),
    INDEX idx_pedidos_fecha (created_at),
    INDEX idx_pedidos_tipo (tipo),
    INDEX idx_pedidos_pagado (pagado),
    INDEX idx_pedidos_tipo_entrega (tipo_entrega),
    INDEX idx_pedidos_fecha_pago (fecha_pago)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;


-- ============================================================
-- TABLA: pedidos_cliente
-- ============================================================
-- Propósito: Pedidos realizados desde la carta web por clientes.
-- Uso: Cliente (crear), admin (verificar pago).
-- Origen de datos: Operación del cliente en la web.
-- Diferencia con pedidos: requieren confirmación de pago del admin
--                         antes de generar una venta.
-- Relación: cliente_id -> clientes, confirmado_por -> usuarios.
-- ============================================================
CREATE TABLE pedidos_cliente (
    id INT PRIMARY KEY AUTO_INCREMENT,
    cliente_id INT DEFAULT NULL,
    cliente_nombre VARCHAR(100) NOT NULL,
    cliente_telefono VARCHAR(15) DEFAULT NULL,
    cliente_direccion VARCHAR(255) DEFAULT NULL,
    cliente_referencia VARCHAR(255) DEFAULT NULL,
    items JSON NOT NULL,
    subtotal DECIMAL(10,2) DEFAULT 0.00,
    igv DECIMAL(10,2) DEFAULT 0.00,
    total DECIMAL(10,2) NOT NULL,
    tipo_entrega ENUM('local', 'delivery', 'paraLlevar', 'motorizada') DEFAULT 'delivery',
    metodo_pago ENUM('efectivo', 'tarjeta', 'yape', 'plin', 'transferencia') NOT NULL,
    tipo_transferencia ENUM('qr', 'maquina') DEFAULT NULL,
    comprobante_pago VARCHAR(255) DEFAULT NULL,
    numero_operacion VARCHAR(50) DEFAULT NULL,
    estado ENUM('pendiente', 'preparando', 'listo', 'entregado', 'cancelado') DEFAULT 'pendiente',
    pagado BOOLEAN DEFAULT FALSE,
    fecha_pago TIMESTAMP NULL DEFAULT NULL,
    confirmado_por INT DEFAULT NULL,
    fecha_confirmacion TIMESTAMP NULL DEFAULT NULL,
    motivo_rechazo VARCHAR(255) DEFAULT NULL,
    venta_id INT DEFAULT NULL,
    observaciones TEXT DEFAULT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    deleted_at TIMESTAMP NULL DEFAULT NULL,

    FOREIGN KEY (cliente_id) REFERENCES clientes(id) ON DELETE SET NULL,
    FOREIGN KEY (confirmado_por) REFERENCES usuarios(id) ON DELETE SET NULL,

    INDEX idx_pedidos_cliente_cliente (cliente_id),
    INDEX idx_pedidos_cliente_estado (estado),
    INDEX idx_pedidos_cliente_pagado (pagado),
    INDEX idx_pedidos_cliente_fecha (created_at),
    INDEX idx_pedidos_cliente_metodo (metodo_pago),
    INDEX idx_pedidos_cliente_tipo (tipo_entrega),
    INDEX idx_pedidos_cliente_transf (tipo_transferencia),
    INDEX idx_pedidos_cliente_confirmado_por (confirmado_por),
    INDEX idx_pedidos_cliente_venta (venta_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;


-- ============================================================
-- TABLA: ventas
-- ============================================================
-- Propósito: Registro contable de ventas concretadas.
-- Uso: Admin (reportes), cajero (cobros).
-- Origen de datos: Se genera automáticamente al confirmar
--                  el pago de un pedido o pedido_cliente.
-- Diferencia con pedidos: es el registro final de contabilidad.
-- ============================================================
CREATE TABLE ventas (
    id INT PRIMARY KEY AUTO_INCREMENT,
    pedido_id INT DEFAULT NULL,
    pedido_cliente_id INT DEFAULT NULL,
    usuario_id INT NOT NULL,
    mesa_id INT DEFAULT NULL,
    cliente_id INT DEFAULT NULL,
    cliente_nombre VARCHAR(100) DEFAULT NULL,
    items JSON NOT NULL,
    subtotal DECIMAL(10,2) DEFAULT 0.00,
    igv DECIMAL(10,2) DEFAULT 0.00,
    descuento DECIMAL(10,2) DEFAULT 0.00,
    total DECIMAL(10,2) NOT NULL,
    metodo_pago ENUM('efectivo', 'tarjeta', 'yape', 'plin', 'transferencia') NOT NULL,
    numero_operacion VARCHAR(50) DEFAULT NULL,
    tipo_entrega ENUM('local', 'delivery', 'paraLlevar', 'motorizada') DEFAULT 'local',
    origen ENUM('mesero', 'cajero', 'admin', 'pedido_web') DEFAULT 'mesero',
    estado ENUM('completada', 'cancelada', 'pendiente') DEFAULT 'completada',
    observaciones TEXT DEFAULT NULL,
    fecha_venta TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    deleted_at TIMESTAMP NULL DEFAULT NULL,

    FOREIGN KEY (pedido_id) REFERENCES pedidos(id) ON DELETE SET NULL,
    FOREIGN KEY (pedido_cliente_id) REFERENCES pedidos_cliente(id) ON DELETE SET NULL,
    FOREIGN KEY (usuario_id) REFERENCES usuarios(id) ON DELETE RESTRICT,
    FOREIGN KEY (mesa_id) REFERENCES mesas(id) ON DELETE SET NULL,
    FOREIGN KEY (cliente_id) REFERENCES clientes(id) ON DELETE SET NULL,

    INDEX idx_ventas_usuario (usuario_id),
    INDEX idx_ventas_pedido (pedido_id),
    INDEX idx_ventas_pedido_cliente (pedido_cliente_id),
    INDEX idx_ventas_mesa (mesa_id),
    INDEX idx_ventas_cliente (cliente_id),
    INDEX idx_ventas_fecha (fecha_venta),
    INDEX idx_ventas_metodo_pago (metodo_pago),
    INDEX idx_ventas_estado (estado),
    INDEX idx_ventas_origen (origen),
    INDEX idx_ventas_tipo_entrega (tipo_entrega),
    INDEX idx_ventas_total (total)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;


-- ============================================================
-- TABLA: historial_actividad
-- ============================================================
-- Propósito: Registro de eventos de negocio (login, logout,
--            registro, creación y cambios de pedidos).
-- Uso: Admin (auditoría), sistema (análisis de uso).
-- Origen de datos: Backend, en cada evento relevante.
-- Diferencia con logs: enfoque de negocio, no técnico.
-- ============================================================
CREATE TABLE historial_actividad (
    id INT PRIMARY KEY AUTO_INCREMENT,
    usuario_id INT DEFAULT NULL,
    cliente_id INT DEFAULT NULL,
    tipo_usuario ENUM('usuario', 'cliente', 'anonimo') DEFAULT 'usuario',
    accion ENUM(
        'login_exitoso',
        'login_fallido',
        'registro',
        'logout',
        'pedido_creado',
        'pedido_completado',
        'pedido_cancelado',
        'navegacion'
    ) NOT NULL,
    descripcion VARCHAR(255) DEFAULT NULL,
    ip VARCHAR(45) DEFAULT NULL,
    user_agent TEXT DEFAULT NULL,
    datos JSON DEFAULT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (usuario_id) REFERENCES usuarios(id) ON DELETE SET NULL,
    FOREIGN KEY (cliente_id) REFERENCES clientes(id) ON DELETE SET NULL,

    INDEX idx_hist_usuario (usuario_id),
    INDEX idx_hist_cliente (cliente_id),
    INDEX idx_hist_tipo (tipo_usuario),
    INDEX idx_hist_accion (accion),
    INDEX idx_hist_fecha (created_at)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;


-- ============================================================
-- TABLA: configuracion
-- ============================================================
-- Propósito: Parámetros generales de la empresa (RUC, dirección,
--            horarios, IGV, redes sociales, etc.).
-- Uso: Admin (consulta), frontend (valores por defecto).
-- Origen de datos: Carga inicial.
-- ============================================================
CREATE TABLE configuracion (
    id INT PRIMARY KEY AUTO_INCREMENT,
    clave VARCHAR(50) UNIQUE NOT NULL,
    valor TEXT DEFAULT NULL,
    tipo ENUM('texto', 'numero', 'booleano', 'json') DEFAULT 'texto',
    descripcion VARCHAR(255) DEFAULT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    deleted_at TIMESTAMP NULL DEFAULT NULL,

    INDEX idx_configuracion_clave (clave),
    INDEX idx_configuracion_tipo (tipo)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;


-- ============================================================
-- TABLA: logs
-- ============================================================
-- Propósito: Auditoría técnica de operaciones sobre la base de datos
--            (INSERT, UPDATE, DELETE en cualquier tabla).
-- Uso: Backend (debug, auditoría técnica), admin.
-- Origen de datos: Backend, en cada operación de escritura.
-- Diferencia con historial_actividad: enfoque técnico.
-- ============================================================
CREATE TABLE logs (
    id INT PRIMARY KEY AUTO_INCREMENT,
    usuario_id INT DEFAULT NULL,
    accion VARCHAR(100) NOT NULL,
    tabla VARCHAR(50) DEFAULT NULL,
    registro_id INT DEFAULT NULL,
    datos_anteriores JSON DEFAULT NULL,
    datos_nuevos JSON DEFAULT NULL,
    ip VARCHAR(45) DEFAULT NULL,
    user_agent TEXT DEFAULT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (usuario_id) REFERENCES usuarios(id) ON DELETE SET NULL,

    INDEX idx_logs_usuario (usuario_id),
    INDEX idx_logs_accion (accion),
    INDEX idx_logs_tabla (tabla),
    INDEX idx_logs_fecha (created_at),
    INDEX idx_logs_registro (tabla, registro_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;


-- ============================================================
-- TABLA: historial_precios
-- ============================================================
-- Propósito: Registro específico de cambios de precio en productos.
-- Uso: Admin (auditoría de precios), reportes de variación.
-- Origen de datos: Backend, al actualizar el precio de un producto.
-- Nota: incluye diferencia y porcentaje de cambio calculados.
-- ============================================================
CREATE TABLE historial_precios (
    id INT PRIMARY KEY AUTO_INCREMENT,
    producto_id INT NOT NULL,
    producto_nombre VARCHAR(100) NOT NULL,
    usuario_id INT DEFAULT NULL,
    usuario_nombre VARCHAR(150) DEFAULT NULL,
    precio_anterior DECIMAL(10,2) NOT NULL,
    precio_nuevo DECIMAL(10,2) NOT NULL,
    diferencia DECIMAL(10,2) AS (precio_nuevo - precio_anterior) STORED,
    porcentaje_cambio DECIMAL(6,2) AS (
        CASE
            WHEN precio_anterior > 0
            THEN ROUND(((precio_nuevo - precio_anterior) / precio_anterior) * 100, 2)
            ELSE 0
        END
    ) STORED,
    motivo VARCHAR(255) DEFAULT NULL,
    ip VARCHAR(45) DEFAULT NULL,
    user_agent TEXT DEFAULT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (producto_id) REFERENCES productos(id) ON DELETE CASCADE,
    FOREIGN KEY (usuario_id) REFERENCES usuarios(id) ON DELETE SET NULL,

    INDEX idx_hist_precio_producto (producto_id),
    INDEX idx_hist_precio_usuario (usuario_id),
    INDEX idx_hist_precio_fecha (created_at)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;


-- ============================================================
-- TABLA: tokens_blacklist
-- ============================================================
-- Propósito: Almacena tokens JWT revocados por logout o
--            invalidación manual.
-- Uso: Middleware de autenticación (verifyToken).
-- Origen de datos: Backend, al procesar logout.
-- ============================================================
CREATE TABLE tokens_blacklist (
    id INT PRIMARY KEY AUTO_INCREMENT,
    token VARCHAR(500) NOT NULL,
    usuario_id INT DEFAULT NULL,
    expiracion TIMESTAMP NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (usuario_id) REFERENCES usuarios(id) ON DELETE CASCADE,

    INDEX idx_tokens_token (token(255)),
    INDEX idx_tokens_usuario (usuario_id),
    INDEX idx_tokens_expiracion (expiracion)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
